10export function tabsReducer(state: TabsState, action: TabsAction): TabsState {
11 switch (action.type) {
12 case 'tabs/open':
13 if (state.tabs.some((tab) => tab.relPath === action.tab.relPath)) {
14 const existing = state.tabs.find((tab) => tab.relPath === action.tab.relPath);
15 return { ...state, activeId: existing?.id ?? null };
16 }
17 return { ...state, tabs: [...state.tabs, action.tab], activeId: action.tab.id };
18 case 'tabs/close':
19 return { ...state, tabs: state.tabs.filter((tab) => tab.id !== action.id) };
20 default:
21 return state;
22 }
